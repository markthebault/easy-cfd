cmake_minimum_required(VERSION 3.16)
project(vvflow_browser LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fexceptions -Wno-unknown-pragmas -include cstdint -include cstdio -include stdexcept")
set(LUA ${CMAKE_CURRENT_SOURCE_DIR}/build/lua-5.2.4/src)
set(CLAPACK ${CMAKE_CURRENT_SOURCE_DIR}/build/clapack-3.2.1-CMAKE)
file(GLOB LUA_SOURCES ${LUA}/*.c)
list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac)\\.c$")
add_library(lua STATIC ${LUA_SOURCES})
target_include_directories(lua PUBLIC ${LUA})
set(CORE vendor/libvvhd/src)
add_library(vvhd STATIC ${CORE}/TEval.cpp ${CORE}/TBody.cpp build/TSpace.cpp ${CORE}/TSortedTree.cpp build/TMatrix.cpp ${CORE}/MConvectiveFast.cpp ${CORE}/MDiffusiveFast.cpp build/MEpsilonFast.cpp ${CORE}/MFlowmove.cpp)
target_include_directories(vvhd PUBLIC vendor/libvvhd/headers port)
target_link_libraries(vvhd PUBLIC lua)
if(EMSCRIPTEN)
  set(LAPACK_SOURCES)
  foreach(name dgesv dgetrf dgetf2 dgetrs dgetri dlaswp dtrtri dtrti2 ilaenv iparmq ieeeck)
    list(APPEND LAPACK_SOURCES ${CLAPACK}/SRC/${name}.c)
  endforeach()
  foreach(name dgemm dgemv dger dscal idamax dswap dtrsm dtrmm dtrmv dtrsv dcopy lsame)
    list(APPEND LAPACK_SOURCES ${CLAPACK}/BLAS/SRC/${name}.c)
  endforeach()
  foreach(name s_cmp i_nint)
    list(APPEND LAPACK_SOURCES ${CLAPACK}/F2CLIBS/libf2c/${name}.c)
  endforeach()
  add_library(linalg STATIC ${LAPACK_SOURCES} port/blas_bridge.c)
  target_include_directories(linalg PRIVATE ${CLAPACK}/INCLUDE port)
  target_link_libraries(vvhd PUBLIC linalg)
else()
  target_link_libraries(vvhd PUBLIC lapack blas)
endif()
add_executable(vvflow browser.cpp)
target_link_libraries(vvflow PRIVATE vvhd)
if(EMSCRIPTEN)
  set_target_properties(vvflow PROPERTIES SUFFIX ".mjs")
  target_link_options(vvflow PRIVATE -fexceptions -sDISABLE_EXCEPTION_CATCHING=0 -sMODULARIZE=1 -sEXPORT_ES6=1 -sENVIRONMENT=web,worker,node -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=33554432 -sMAXIMUM_MEMORY=1073741824 -sEXPORTED_FUNCTIONS=['_vv_init','_vv_step','_vv_count','_vv_particles','_vv_bodies','_vv_body_count','_vv_stats','_vv_error','_vv_dispose','_vv_density','_vv_scene','_malloc','_free'] -sEXPORTED_RUNTIME_METHODS=['HEAPF64','UTF8ToString'] -sFILESYSTEM=0)
endif()
